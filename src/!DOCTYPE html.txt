  <script>
    // Define a data no topo
    document.getElementById('dataAtual').innerText = new Date().toLocaleDateString('pt-BR', { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' });

    // Função de Login conectada ao seu Código.gs
    function fazerLogin() {
      const senha = document.getElementById('inputSenha').value;
      
      // UX visual de carregamento
      document.getElementById('btnText').innerText = "Verificando...";
      document.getElementById('btnSpinner').classList.remove('hidden');
      document.getElementById('msgLogin').classList.add('hidden');

      // Chama a função verificarAcesso do seu Código.gs
      google.script.run
        .withSuccessHandler(function(resposta) {
          if(resposta.sucesso) {
            // Esconde login, mostra app
            document.getElementById('loginScreen').classList.add('hidden');
            document.getElementById('appScreen').classList.remove('hidden');
            
            // Inicia o carregamento dos dados reais
            carregarDadosIniciais(senha);
          } else {
            // Falha no login
            document.getElementById('msgLogin').classList.remove('hidden');
            resetarBotaoLogin();
          }
        })
        .withFailureHandler(function(erro) {
            alert("Erro de conexão: " + erro);
            resetarBotaoLogin();
        })
        .verificarAcesso(senha);
    }

    function resetarBotaoLogin() {
        document.getElementById('btnText').innerText = "Acessar Sistema";
        document.getElementById('btnSpinner').classList.add('hidden');
        document.getElementById('inputSenha').value = "";
    }

    function sair() {
        document.getElementById('appScreen').classList.add('hidden');
        document.getElementById('loginScreen').classList.remove('hidden');
        document.getElementById('inputSenha').value = "";
    }

    // Função para buscar os dados usando sua função buscarDadosBackend
    function carregarDadosIniciais(senha) {
        google.script.run
        .withSuccessHandler(function(dados) {
            console.log("Dados carregados com sucesso!", dados);
            // Aqui futuramente vamos popular as tabelas.
            // O sistema visual já está montado aguardando os formulários de Receita/Despesa.
        })
        .withFailureHandler(function(erro) {
            alert("Erro ao puxar dados: " + erro.message);
        })
        .buscarDadosBackend(senha);
    }
  </script>
